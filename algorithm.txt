atlases[] = [nearest, linear];

foreach(t in textures.t)
{
  if(exist t.image)
    ss = new(t.image) { LID=t.ID GID=next }
    atlases[t.filter].add(ss)
  else
    ss = new(t.metadata, maxRect=textures.t.count) { LID=t.ID GID=next }
    atlases[t.filter].add(ss)
}

refs = []
foreach(assetPos in allAssetPos) 
{
  a = assetPos.a
  if(a ?? 0 <= lastOrigsName)
    continue

  re = textures.re[a]
  if(re == null)
    throw invalid s reference a=...
  
  atlas = atlases.FirstOrDefault(atlas => atlas.Contains(re.t))
  if(atlas == null)
    throw s maps to unknown ss
  ss = atlas[re.t] // this is never null

  if(ss.HasImage)
  {
    s = new(ss, w, h, x, y) { LN=next, LID=next, GN=next, GID=next }
    ss.add(s)
    refs.add(new(assetPos, s))
    // debug: render s image as PNG?
  }
  else
  {
    if(!re.HasImage)
      throw no ss image nor s image could be found for s region = ...
  
    s = new(tex.Image) { x, y }
    ss.add(s) { LN=next, LID=next, GN=next, GID=next }
    refs.add(assetPos, s)
  }
}

parallel.foreach(ss in atlases.SelectMany(atlas => atlas.ss))
{
  if(!ss.hasImage)
    ss.render()
  ss.ExportToCim()
  // debug: ss.ExportToPng()
}
